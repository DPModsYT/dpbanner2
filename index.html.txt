        function renderSavedUrls() {

            if (savedConns.length === 0) { container.style.display = 'none'; return; }

                const displayHost = conn.url.replace(/^https?:\/\//, '').replace(/\/+$/, '');
                list.innerHTML += `

                        <div style="flex:1; overflow:hidden;">
                            <div style="font-size:0.82rem; font-weight:800; color:#fff; text-overflow:ellipsis; overflow:hidden; white-space:nowrap;"><i class="fas fa-database" style="color:var(--cyan); margin-right:8px;"></i>${displayHost}</div>

                        <div style="background:var(--danger-dim); color:var(--danger); width:30px; height:30px; border-radius:8px; display:flex; justify-content:center; align-items:center;" onclick="removeSavedUrl(${index}, event)"><i class="fas fa-trash" style="font-size:0.75rem;"></i></div>

            });

        async function login() {
            const btn = document.getElementById('login-btn');
            btn.innerHTML = `<i class="fas fa-spinner fa-spin"></i> CONNECTING...`;
            
            let url = document.getElementById('auth-url').value.trim();
            const secret = document.getElementById('auth-secret').value.trim();

            if (!url || !secret) {
                showIsland("Enter URL and Token!", "error");
                btn.innerHTML = "CONNECT DATABASE"; return;
            }

            url = url.replace(/\/+$/, '').replace(/\.json$/, '');
            DB_URL = url;
            DB_SECRET = secret;

            try {
                const res = await fetch(`${DB_URL}/.json?auth=${DB_SECRET}`);
                if (!res.ok) throw new Error("Auth Failed");
                
                const data = await res.json() || {};
                DB = {
                    App_Status: data.App_Status || {},
                    Banned_HWIDs: data.Banned_HWIDs || {},
                    Keys: data.Keys || {},
                    Trials: data.Trials || {}
                };
                lastSyncedKeysHash = JSON.stringify(DB.Keys);

                setStorage('dp_panel_url', DB_URL);
                setStorage('dp_panel_secret', DB_SECRET);

                let savedConns = JSON.parse(getStorage('dp_panel_conns', '[]')) || [];
                const existingIndex = savedConns.findIndex(c => c.url === DB_URL);
                if (existingIndex !== -1) savedConns.splice(existingIndex, 1);
                savedConns.unshift({ url: DB_URL, secret: DB_SECRET });
                if (savedConns.length > 5) savedConns.pop(); 
                setStorage('dp_panel_conns', JSON.stringify(savedConns));
                
                const loginScreen = document.getElementById('login-screen');
                loginScreen.style.opacity = '0';
                loginScreen.style.pointerEvents = 'none';
                
                setTimeout(() => {
                    loginScreen.style.display = 'none';
                    document.getElementById('main-app').style.display = 'flex'; 
                    document.getElementById('bottom-nav').style.display = 'flex';
                    
                    loadConfig(); 
                    
                    currentFilter = 'all';
                    visibleKeysCount = PAGE_CHUNK;
                    document.querySelectorAll('.filter-chip').forEach(c => c.classList.remove('active'));
                    document.getElementById('chip-all').classList.add('active'); 
                    
                    renderAll();
                    showIsland("Connected to DPMods Panel", "success", "fa-check-circle");
                }, 300);

                if (window.syncInterval) clearInterval(window.syncInterval);
                window.syncInterval = setInterval(silentBackgroundSync, 10000);

            } catch (e) {
                showIsland("Connection Failed! Check URL/Token", "error");
                btn.innerHTML = "CONNECT DATABASE";
            }
        }

        async function silentBackgroundSync() {
            try {
                const res = await fetch(`${DB_URL}/Keys.json?auth=${DB_SECRET}`);
                if (res.ok) {
                    const keysData = await res.json() || {};
                    const newHash = JSON.stringify(keysData);
                    if (newHash !== lastSyncedKeysHash) {
                        DB.Keys = keysData;
                        lastSyncedKeysHash = newHash;
                        renderAll();
                    }
                }
            } catch (e) {}
        }

            document.documentElement.style.setProperty('--border-cyan', `rgba(${r}, ${g}, ${b}, 0.35)`);
            document.documentElement.style.setProperty('--cyan-glow', `0 0 15px rgba(${r}, ${g}, ${b}, 0.4)`);
        }

        function getDeviceEntries(data) {
            if (!data || !data.Devices || typeof data.Devices !== 'object') return [];
            return Object.entries(data.Devices);
        }

        function getKeyStatus(data) {
            if (data.Banned === true) return { id: 'banned', class: 'banned', text: 'BANNED', color: 'var(--danger)' };
            
            if (data.ExpiryDate) {
                const exp = new Date(data.ExpiryDate);
                exp.setHours(23, 59, 59, 999);
                if (new Date() > exp) {
                    return { id: 'expired', class: 'expired', text: 'EXPIRED', color: 'var(--warning)' };
                }
            }

            const devCount = getDeviceEntries(data).length;
            if (devCount === 0) return { id: 'unused', class: 'unused', text: 'UNUSED', color: 'var(--text-muted)' };
            return { id: 'active', class: 'active', text: 'ACTIVE', color: 'var(--success)' };
        }

        function renderStats() {
            let total = 0, bound = 0, expired = 0;
            Object.values(DB.Keys || {}).forEach(k => {
                if (!k || typeof k !== 'object') return;
                total++;
                bound += getDeviceEntries(k).length;

        function renderKeys() {
            const list = document.getElementById('keys-list');
            const loadMoreBox = document.getElementById('keys-load-more-box');
            const search = document.getElementById('search-keys').value.toLowerCase().trim();
            
            const rawKeys = Object.entries(DB.Keys || {}).reverse();
            
            const filtered = [];
            for (let i = 0; i < rawKeys.length; i++) {
                const [keyId, data] = rawKeys[i];
                if (!data || typeof data !== 'object') continue;
                
                const statusObj = getKeyStatus(data);
                if (currentFilter !== 'all' && currentFilter !== statusObj.id) continue;

                const devEntries = getDeviceEntries(data);
                if (search) {
                    let match = keyId.toLowerCase().includes(search);
                    if (!match && data.ExpiryDate && data.ExpiryDate.toLowerCase().includes(search)) match = true;
                    if (!match && devEntries.length > 0) {
                        match = devEntries.some(([hwid]) => hwid.toLowerCase().includes(search));
                    }
                    if (!match) continue;
                }
                
                filtered.push({ keyId, data, statusObj, devEntries });
            }

            if (filtered.length === 0) {
                list.innerHTML = `<div class="empty-state">NO KEYS FOUND</div>`;
                loadMoreBox.innerHTML = '';
                return;
            }

            const sliceToRender = filtered.slice(0, visibleKeysCount);
            let htmlBuffer = "";

            for (let i = 0; i < sliceToRender.length; i++) {
                const { keyId, data, statusObj, devEntries } = sliceToRender[i];
                const isExpanded = expandedKey === keyId;
                const devCount = devEntries.length;
                
                let devListHtml = "";
                if (devCount > 0) {
                    devListHtml = `<div class="dev-list">
                        <div class="dev-list-header">
                            <span>BOUND DEVICES (${devCount}/${data.DeviceLimit || 1})</span>
                            <span>TAP TO COPY</span>
                        </div>`;
                    devEntries.forEach(([hwid, dVal]) => {
                        const actualId = (dVal && typeof dVal === 'object' && dVal.DeviceID) ? dVal.DeviceID : hwid;
                        devListHtml += `
                            <div class="dev-item">
                                <span class="dev-hwid" onclick="copyToClipboard('${actualId}', event)">
                                    <i class="fas fa-microchip" style="color:var(--text-muted); margin-right:6px; font-size:0.7rem;"></i>${actualId}
                                </span>
                                <i class="fas fa-copy" style="color:var(--text-muted); font-size:0.72rem; cursor:pointer;" onclick="copyToClipboard('${actualId}', event)"></i>
                            </div>
                        `;

        function openBanModal() {
            document.getElementById('mod-ban-id').value = '';
            document.getElementById('ban-modal').classList.add('active');
        }
        
        async function saveBan() {
            const hwid = document.getElementById('mod-ban-id').value.trim();
            if (!hwid) return showIsland("Enter HWID!", "error");
            if (!DB.Banned_HWIDs) DB.Banned_HWIDs = {};
            DB.Banned_HWIDs[hwid] = "true"; 
            await syncNode(`Banned_HWIDs/${encodeURIComponent(hwid)}`, "true");
            closeModals(); renderAll(); showIsland("HWID Banned", "success", "fa-skull");
        }

        async function unbanHWID(hwid) {
            if (confirm(`Unban ${hwid}?`)) {
                delete DB.Banned_HWIDs[hwid]; 
                await syncNode(`Banned_HWIDs/${encodeURIComponent(hwid)}`, null);
                renderAll(); showIsland("Unbanned Successfully", "success");
            }
        }

        function renderBans() {
            const list = document.getElementById('bans-list');
            const search = document.getElementById('search-bans').value.toLowerCase().trim();
            const banObj = DB.Banned_HWIDs || {};
            const entries = Object.entries(banObj);
            
            let html = "";

        /* --- TRIALS LOGIC --- */
        function parseTimestamp(ts) {
            if (!ts || Number(ts) <= 0) return 0;
            const num = Number(ts);
            return num < 10000000000 ? num * 1000 : num;
        }

        function renderTrials() {
            const list = document.getElementById('trials-list');
            const search = document.getElementById('search-trials').value.toLowerCase().trim();
            const trialsObj = DB.Trials || {};
            const entries = Object.entries(trialsObj);

            let html = "";
            const nowMs = Date.now();

            for (let i = 0; i < entries.length; i++) {
                const [hwid, val] = entries[i];
                if (search && !hwid.toLowerCase().includes(search)) continue;
                
                let endMs = 0;
                if (typeof val === 'object' && val !== null) {
                    endMs = parseTimestamp(val.EndTime || val.expiry || val.StartTime || 0);
                } else {
                    endMs = parseTimestamp(val);
                }

                const isExpired = endMs > 0 && nowMs > endMs;
                const statusColor = isExpired ? "var(--warning)" : "var(--success)";
                const statusText = isExpired ? "TRIAL EXPIRED" : "ACTIVE TRIAL";
                const timeStr = endMs > 0 ? new Date(endMs).toLocaleString() : "Active";

                html += `
                    <div class="ios-card" style="padding:16px; margin-bottom:10px; border-left:4px solid ${statusColor};">
                        <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:6px;">
                            <div style="font-family:'JetBrains Mono', monospace; font-weight:800; color:#fff; font-size:0.92rem; cursor:pointer; word-break:break-all;" onclick="copyToClipboard('${hwid}', event)">
                                ${hwid}
                            </div>
                            <button class="sq-btn" style="background:var(--danger-dim); color:var(--danger); border:none; width:38px; height:38px;" onclick="revokeTrial('${hwid}')"><i class="fas fa-trash"></i></button>
                        </div>
                        <div style="display:flex; justify-content:space-between; font-size:0.72rem; font-weight:700;">
                            <span style="color:${statusColor}; letter-spacing:0.8px;">${statusText}</span>
                            <span style="color:var(--text-muted); font-family:'JetBrains Mono', monospace;">${timeStr}</span>
                        </div>
                    </div>`;
            }
            list.innerHTML = html || `<div class="empty-state">No Active Trials</div>`;
        }

        async function revokeTrial(hwid) {
            if (confirm(`Revoke Trial for ${hwid}?`)) {
                delete DB.Trials[hwid]; 
                await syncNode(`Trials/${encodeURIComponent(hwid)}`, null);
                renderAll(); showIsland("Trial Revoked", "success");
            }
        }

        let islandTimeout;
        function showIsland(msg, type = "success", customIcon = null) {
            const island = document.getElementById('dynamic-island');
            island.innerHTML = `<i class="fas ${customIcon || (type === 'success' ? 'fa-check-circle' : 'fa-exclamation-triangle')}"></i> ${msg}`;
            island.style.color = type === "success" ? "var(--cyan)" : "var(--danger)";
            island.classList.add('show');
            clearTimeout(islandTimeout);
            islandTimeout = setTimeout(() => { island.classList.remove('show'); }, 2600); 
        }
